    static class WowBKnownMerchants
    {
        // __HIGH_MATCH__
        public static bool Living = true;
        public static bool IsRecordedLive(VendorFile.VendorPoint p, WowBObservedUnit u, GreenMagic.Memory? m, WowBReader? r, bool allow70938Candidate = false)
            => m != null && r != null && Living && IsLivingObservedNpc(u, m, r) &&
               (p.MerchantGuids.Contains(u.Guid.ToString()) || (allow70938Candidate && r.IsExperimental70938 && Same70938HighCandidate(p,u.Guid)));
        public static bool IsLivingObservedNpc(WowBObservedUnit u, GreenMagic.Memory? m, WowBReader? r)
            => m != null && r != null && Living && !u.Guid.IsEmpty && !u.IsLocalPlayer && !u.HasPlayerVtable;
        // The closest living non-player unit to the player inside the radius. Proximity is
        // not merchant type: the caller logs the choice and re-reads the same object later.
        public static bool TryFindNearestNpc(WowBWorldSnapshot snap, double maxYards, out WowBObservedUnit npc, out double yards, Func<WowBObservedUnit, bool>? accept = null)
        {
            npc = default; yards = double.MaxValue;
            if (!snap.TryGetLocalPlayer(out var me)) return false;
            foreach (var c in snap.Units)
            {
                if (c.IsLocalPlayer || c.HasPlayerVtable || c.Guid.IsEmpty) continue;
                if (accept != null && !accept(c)) continue;
                double d = ClickAimMath.Len(c.Location.X - me.Location.X, c.Location.Y - me.Location.Y);
                if (d <= 0.4 || d > maxYards || d >= yards) continue;
                if (!IsLivingObservedNpc(c, ObjectManager.Wow, ObjectManager.WowB)) continue;
                yards = d; npc = c;
            }
            return !npc.Guid.IsEmpty;
        }
    }
    static class Fixture
    {
        static bool _stop, _vendorDue, _vendorAutoPending, _vendorManualPending, Fighting, Readable = true;
        static WowBGuid _vendorNearestGuid;
        static WowBReader? _vendorAutoReader; static GreenMagic.Memory? _vendorAutoMemory; static WowBGuid _vendorAutoPlayer;
        static WowBWorldSnapshot snapshot = new();
        static bool TryReadGatherCombat(out bool fighting) { fighting=Fighting; return Readable; }
        static bool Capture(out WowBWorldSnapshot? s) { s=snapshot; return true; }
        static bool AutoVendorIdentityMatches(WowBReader r, GreenMagic.Memory m, WowBGuid g) => true;
        // __PRODUCTION__
        static void Check(bool ok, string why) { if (!ok) throw new Exception(why); }
        public static void Main()
        {
            ObjectManager.WowB = new WowBReader { IsExperimental70938 = true }; ObjectManager.Wow = new GreenMagic.Memory();
            var town = new VendorFile.VendorPoint { TownMap = 1 };
            var saved = new WowBGuid(42, 0x206B500020043D80); town.MerchantGuids.Add(saved.ToString());
            var player = new WowBObservedUnit { Guid = new WowBGuid(1, 1), IsLocalPlayer = true };
            var replacement = new WowBObservedUnit { Guid = new WowBGuid(43, saved.High), UnitAddress = 2, RecordAddress = 20, Location = new Location { X = 4 } };
            snapshot.Player = player; snapshot.Units.Add(player); snapshot.Units.Add(replacement);
            Check(TryFindVendorNpc(town, out var found, out _) && found.Guid == replacement.Guid, "unique respawn candidate found");
            Check(VendorCandidateStillValid(town, found, out _, out _, out _), "candidate identity remains valid");
            var duplicate = replacement; duplicate.Guid = new WowBGuid(44, saved.High); duplicate.UnitAddress=3; duplicate.RecordAddress=30; duplicate.Location.X=10;
            snapshot.Units.Add(duplicate);
            // Ambiguous same-high candidates are never substituted, but the nearest living
            // NPC is still used: an unidentifiable merchant is better than no trip at all.
            _vendorNearestGuid = default;
            Check(TryFindVendorNpc(town, out var nearestPick, out _) && nearestPick.Guid == replacement.Guid, "ambiguous same-high candidates fall back to the nearest living NPC");
            Check(VendorCandidateStillValid(town, nearestPick, out _, out _, out _), "nearest fallback identity stays valid while the same object is loaded");
            snapshot.Units.Remove(duplicate);
            var exact = replacement; exact.Guid = saved; exact.UnitAddress=4; exact.RecordAddress=40; exact.Location.X=6; snapshot.Units.Add(exact);
            Check(TryFindVendorNpc(town, out found, out _) && found.Guid == saved, "exact confirmation preferred even farther away");
            exact.Location.X=100; snapshot.Units[2]=exact;
            Check(!TryFindVendorNpc(town, out _, out _), "observed exact prevents candidate substitution");
            snapshot.Units.RemoveAt(2);
            town.TownMap = 2; Check(!TryFindVendorNpc(town, out _, out _), "wrong map"); town.TownMap=1;
            town.X = 100; Check(!TryFindVendorNpc(town, out _, out _), "outside town"); town.X=0;
            // A nearer stranger distinguishes the two paths: only 70938 may substitute a
            // same-high candidate, while every build may use the nearest living NPC.
            var stranger = new WowBObservedUnit { Guid = new WowBGuid(99, 7), UnitAddress = 9, RecordAddress = 90, Location = new Location { X = 2 } };
            snapshot.Units.Add(stranger); _vendorNearestGuid = default;
            ObjectManager.WowB.IsExperimental70938 = false;
            Check(TryFindVendorNpc(town, out var legacyPick, out _) && legacyPick.Guid == stranger.Guid, "legacy builds use the nearest living NPC, never the same-high substitution");
            ObjectManager.WowB.IsExperimental70938 = true; _vendorNearestGuid = default;
            Check(TryFindVendorNpc(town, out var highPick, out _) && highPick.Guid == replacement.Guid, "70938 prefers the unique same-high candidate over a nearer stranger");
            snapshot.Units.Remove(stranger); _vendorNearestGuid = default; WowBKnownMerchants.Living = false; Check(!TryFindVendorNpc(town, out _, out _), "unreadable live candidate rejected"); WowBKnownMerchants.Living=true; _vendorNearestGuid=default;
            Fighting=true; Check(!VendorCandidateStillValid(town, replacement, out _, out _, out _), "combat preempts"); Fighting=false;
            var changed = replacement; changed.RecordAddress++; snapshot.Units[1]=changed;
            Check(!VendorCandidateStillValid(town, replacement, out _, out _, out _), "changed record blocks replay");
            Console.WriteLine("Merchant search/identity fixtures passed with synthetic snapshots.");
        }

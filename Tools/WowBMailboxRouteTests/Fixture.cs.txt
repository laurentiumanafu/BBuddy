    static class RouteRunner
    {
        public static bool StopRequested, PollInterruptDuringWalk, RequireMeshDuringSession, LastRunInterrupted, Result = true, CombatDuringWalk;
        public static Func<bool>? ShouldInterrupt;
        public static bool RunWithinSession(ClickAimMover mover, List<RoutePoint> route, int legs)
        {
            if (!RequireMeshDuringSession || !PollInterruptDuringWalk) throw new Exception("mesh/interrupt not required");
            if (CombatDuringWalk) Fixture.Fighting=true;
            LastRunInterrupted = ShouldInterrupt != null && ShouldInterrupt();
            return Result && !LastRunInterrupted;
        }
    }
    static class Fixture
    {
        static VendorFile.MailboxPoint? _vendorMailboxPending;
        static WowBReader? _vendorMailboxReader;
        static GreenMagic.Memory? _vendorMailboxMemory;
        static WowBGuid _vendorMailboxPlayer;
        static bool _stop; public static bool Fighting; static int _vendorRetryAt;
        static bool AutoVendorIdentityMatches(WowBReader r, GreenMagic.Memory m, WowBGuid g) => g == r.Player.Guid;
        static bool TryReadGatherCombat(out bool fighting) { fighting=Fighting; return true; }
        static void Stop() { _stop=true; ClickAimAbort.Requested=true; }
        // __PRODUCTION__
        static void Check(bool b, string why) { if (!b) throw new Exception(why); }
        static void Reset()
        {
            _stop=false; ClickAimAbort.Requested=false; RouteRunner.StopRequested=false; RouteRunner.LastRunInterrupted=false;
            RouteRunner.Result=true; RouteRunner.CombatDuringWalk=false; Fighting=false;
            _vendorMailboxReader=new WowBReader(); _vendorMailboxMemory=new GreenMagic.Memory(); _vendorMailboxPlayer=new WowBGuid { Value=7 };
            ObjectManager.WowB=_vendorMailboxReader; ObjectManager.Wow=_vendorMailboxMemory;
            _vendorMailboxPending=new VendorFile.MailboxPoint { Map=1, X=10, Y=20, Z=5 };
        }
        public static void Main()
        {
            Reset(); var prior = new Func<bool>(() => false); RouteRunner.ShouldInterrupt=prior;
            Check(WalkToRecordedMailbox(new ClickAimMover()) && _stop && _vendorMailboxPending == null, "arrival stops for manual mail");
            Check(RouteRunner.ShouldInterrupt == prior && !RouteRunner.RequireMeshDuringSession && !RouteRunner.PollInterruptDuringWalk, "restore route controls");
            Reset(); Fighting=true; Check(!WalkToRecordedMailbox(new ClickAimMover()) && !_stop && _vendorMailboxPending != null, "combat yields");
            Fighting=false; Check(WalkToRecordedMailbox(new ClickAimMover()) && _stop, "resume after combat");
            Reset(); RouteRunner.CombatDuringWalk=true; Check(!WalkToRecordedMailbox(new ClickAimMover()) && !_stop && _vendorMailboxPending != null, "interrupt retains stage");
            Reset(); _vendorMailboxReader!.Map=2; Check(!WalkToRecordedMailbox(new ClickAimMover()) && !_stop && _vendorMailboxPending != null, "map mismatch does not move");
            Reset(); _vendorMailboxReader!.Player.Guid=new WowBGuid { Value=8 }; Check(!WalkToRecordedMailbox(new ClickAimMover()) && _stop, "changed character stops");
            Reset(); RouteRunner.Result=false; Check(!WalkToRecordedMailbox(new ClickAimMover()) && _stop, "failed mesh stops");
            Reset(); _vendorMailboxReader!.Player.Location.X=50; Check(!WalkToRecordedMailbox(new ClickAimMover()) && _stop, "route return alone is not arrival");
            Reset(); ClickAimAbort.Requested=true; Check(!WalkToRecordedMailbox(new ClickAimMover()) && _vendorMailboxPending != null, "STOP holds");
            Console.WriteLine("Mailbox route fixtures passed; no live navigation proof.");
        }

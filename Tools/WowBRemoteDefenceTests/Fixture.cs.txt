    public readonly struct WowBGuid
    {
        public readonly int Value;
        public WowBGuid(int value) { Value = value; }
        public bool IsEmpty => Value == 0;
        public static bool operator ==(WowBGuid a, WowBGuid b) => a.Value == b.Value;
        public static bool operator !=(WowBGuid a, WowBGuid b) => !(a == b);
        public override bool Equals(object? obj) => obj is WowBGuid g && this == g;
        public override int GetHashCode() => Value;
        public override string ToString() => Value.ToString();
    }

    public sealed class WowBWorldSnapshot
    {
        public List<WowBObservedUnit> Units = new();
        public WowBReader Reader = null!;
        public bool TryGetLocalPlayer(out WowBObservedUnit found) { found = Units[0]; return true; }
        public bool TryGetUnit(WowBGuid guid, out WowBObservedUnit found)
        { foreach (var u in Units) if (u.Guid == guid) { found = u; return true; } found = default; return false; }
        public bool TryGetSelectedTarget(out WowBObservedUnit found) => TryGetUnit(Reader.Selected, out found);
    }
    public sealed class WowBReader
    {
        public struct WowBUnitFields { public uint Health, UnitFlags; }
        public Dictionary<ulong,WowBUnitFields> Fields = new();
        public WowBGuid Selected; public bool IsExperimental70707 = true, IsExperimental70862 = false;
        public WowBGuid ReadTargetGuid() => Selected;
        public bool TryGetUnitFields(ulong p, out WowBUnitFields f) => Fields.TryGetValue(p, out f);
    }

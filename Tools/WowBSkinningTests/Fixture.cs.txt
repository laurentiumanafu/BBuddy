    internal struct WowBGuid : IEquatable<WowBGuid>
    {
        internal ulong High, Low;
        internal bool IsEmpty => High == 0 && Low == 0;
        internal WowBGuid(uint entry) { High = (ulong)entry << 6; Low = 1; }
        public bool Equals(WowBGuid other) => High == other.High && Low == other.Low;
        public override bool Equals(object? o) => o is WowBGuid g && Equals(g);
        public override int GetHashCode() => HashCode.Combine(High, Low);
        public static bool operator ==(WowBGuid a, WowBGuid b) => a.Equals(b);
        public static bool operator !=(WowBGuid a, WowBGuid b) => !a.Equals(b);
    }

namespace Styx.WoWInternals
{
    readonly struct WowBGuid : IEquatable<WowBGuid>
    {
        private readonly int value;
        public WowBGuid(int value) { this.value = value; }
        public bool IsEmpty => value == 0;
        public bool Equals(WowBGuid other) => value == other.value;
        public override bool Equals(object? other) => other is WowBGuid guid && Equals(guid);
        public override int GetHashCode() => value;
        public static bool operator ==(WowBGuid a, WowBGuid b) => a.Equals(b);
        public static bool operator !=(WowBGuid a, WowBGuid b) => !a.Equals(b);
    }
    struct WowBObservedUnit { public WowBGuid Guid; public ulong UnitAddress, RecordAddress; }
    sealed class WowBWorldSnapshot
    {
        public WowBObservedUnit Player;
        public bool TryGetLocalPlayer(out WowBObservedUnit player) { player = Player; return !player.Guid.IsEmpty; }
    }
    sealed class WowBReader
    {
        public struct WowBUnitFields { public uint Health, MaxHealth, Power; }
        public bool IsExperimental70707 = true, Readable = true;
        public uint Map = 0, Health = 100, MaxHealth = 100, Power = 100;
        public WowBGuid Selected;
        public WowBObservedUnit Player = new WowBObservedUnit { Guid = new WowBGuid(1), UnitAddress = 100, RecordAddress = 200 };
        public bool TryReadMapId(out uint map) { map = Map; return Readable; }
        public bool TryCaptureWorldSnapshot(out WowBWorldSnapshot? snapshot) { snapshot = Readable ? new WowBWorldSnapshot { Player = Player } : null; return Readable; }
        public bool TryGetUnitFields(ulong address, out WowBUnitFields fields) { fields = new WowBUnitFields { Health = Health, MaxHealth = MaxHealth, Power = Power }; return Readable && address == Player.UnitAddress; }
        public WowBGuid ReadTargetGuid() => Selected;
    }
    static class ObjectManager { public static Memory? Wow; public static WowBReader? WowB; }
}
namespace Styx.Logic.Pathing
{
    enum ClickAimRotationClass { Auto, Generic, Druid }
    enum ClickAimSpellRole { Unknown, Buff, Ignore, Damage }
    sealed class ClickAimAction
    {
        public int Id, KeyVk, Slot; public ulong Raw; public string Name = "";
        public bool IsBuff => ClickAimCombat.RoleOf(Id) == ClickAimSpellRole.Buff;
    }
    static class ClickAimSettings { public static bool UseBuffs = true; public static ClickAimRotationClass RotationClass = ClickAimRotationClass.Auto; }
    static class ClickAimClassRotation { public static ClickAimRotationClass Resolve(ClickAimRotationClass type, List<ClickAimAction> bar) => type; }
    static class ClickAimCombat
    {
        public const int CastSettleMs = 1800;
        public static List<ClickAimAction> Bar = new();
        public static Dictionary<int, ClickAimSpellRole> Overrides = new();
        public static List<ClickAimAction> ReadBar(WowBReader reader) => new List<ClickAimAction>(Bar);
        public static string NameOf(int id) => id switch { 1 => "Arcane Intellect", 2 => "Frost Armor", 3 => "Thorns", 4 => "Cat Form", 20550 => "Endurance", _ => "spell" };
        public static ClickAimSpellRole RoleOverrideOf(int id) => Overrides.TryGetValue(id, out var role) ? role : ClickAimSpellRole.Unknown;
        public static ClickAimSpellRole RoleOf(int id) => Overrides.TryGetValue(id, out var role) ? role : ClickAimSpellRole.Buff;
    }
    static class WowBGrindFactions
    {
        public static bool BarSafe = true, Ghost, GhostReadable = true;
        public static bool Experimental70707BarSafe(WowBReader reader) => BarSafe;
        public static bool TryReadGhostFlagCandidate(Memory m, WowBReader r, WowBObservedUnit p, out bool ghost) { ghost = Ghost; return GhostReadable; }
    }
    static class RouteRunner { public static bool StopRequested; }
    static class GrindRunner { public static bool Combat, CombatReadable = true; public static bool TryReadGatherCombat(out bool fighting) { fighting = Combat; return CombatReadable; } }
    static class ClickAimAbort { public static bool Requested; public static bool Wait(int ms) => Requested; }
    static class ClickAimInput
    {
        public static bool F1Works = true, StopOnF1, StopAfterFirst, DisableAfterFirst, CombatAfterFirst, ChangeSecondAfterFirst, ChangeTargetAfterFirst, ChangeMapAfterFirst;
        public static List<int> Keys = new();
        public static void TapKey(IntPtr hwnd, int key, int ms)
        {
            Keys.Add(key);
            var reader = ObjectManager.WowB!;
            if (key == 0x70) { if (F1Works) reader.Selected = reader.Player.Guid; if (StopOnF1) ClickAimAbort.Requested = true; }
            else if (key == 0x31)
            {
                if (CombatAfterFirst) GrindRunner.Combat = true;
                if (StopAfterFirst) ClickAimAbort.Requested = true;
                if (DisableAfterFirst) ClickAimSettings.UseBuffs = false;
                if (ChangeSecondAfterFirst) ClickAimCombat.Bar[1] = Harness.Action(999, 2);
                if (ChangeTargetAfterFirst) reader.Selected = new WowBGuid(9);
                if (ChangeMapAfterFirst) reader.Map = 1;
            }
        }
    }
    static class Harness
    {
        public static ClickAimAction Action(int id, int slot) => new ClickAimAction { Id = id, Slot = slot, KeyVk = 0x30 + slot, Raw = (ulong)id, Name = ClickAimCombat.NameOf(id) };
        private static void Check(bool value, string label) { if (!value) throw new Exception(label); }
        private static void Reset()
        {
            ObjectManager.Wow = new Memory(); ObjectManager.WowB = new WowBReader { Selected = new WowBGuid(8) };
            ClickAimCombat.Bar = new List<ClickAimAction> { Action(1, 1), Action(2, 2), Action(3, 3) };
            ClickAimCombat.Overrides.Clear(); ClickAimInput.Keys.Clear();
            ClickAimSettings.UseBuffs = true; ClickAimSettings.RotationClass = ClickAimRotationClass.Auto;
            GrindRunner.Combat = ClickAimAbort.Requested = RouteRunner.StopRequested = false;
            GrindRunner.CombatReadable = WowBGrindFactions.BarSafe = WowBGrindFactions.GhostReadable = ClickAimInput.F1Works = true;
            WowBGrindFactions.Ghost = ClickAimInput.StopOnF1 = ClickAimInput.StopAfterFirst = ClickAimInput.DisableAfterFirst = ClickAimInput.CombatAfterFirst = ClickAimInput.ChangeSecondAfterFirst = ClickAimInput.ChangeTargetAfterFirst = ClickAimInput.ChangeMapAfterFirst = false;
        }
        public static void Main()
        {
            long now = 0; int stoppedMovement = 0;
            Func<bool> stopped = () => false;
            Action stopMovement = () => stoppedMovement++;
            var controller = new ClickAimBuffController(() => now);
            Reset();
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && stoppedMovement == 1 &&
                  ClickAimInput.Keys.Count == 4 && ClickAimInput.Keys[0] == 0x70 && ClickAimInput.Keys[1] == 0x31 &&
                  ClickAimInput.Keys[2] == 0x32 && ClickAimInput.Keys[3] == 0x33, "fresh session selects self and attempts all three buffs");
            ClickAimInput.Keys.Clear(); now = ClickAimBuffSchedule.RefreshMs - 5000;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "no early refresh");
            now = ClickAimBuffSchedule.RefreshMs;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 3, "30-minute refresh, existing self-target preserved");
            ClickAimInput.Keys.Clear(); now++;
            controller.Reset();
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 3, "confirmed resurrection resets timer immediately");
            Reset(); controller.Reset(); now = 0; GrindRunner.Combat = true;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "combat blocks targeting and buffs");
            GrindRunner.Combat = false; now = 5000;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 4, "deferred start resumes out of combat");
            Reset(); controller.Reset(); now = 0;
            Check(!controller.TryApply(new IntPtr(7), stopped, () => throw new Exception("stop failed")) && ClickAimInput.Keys.Count == 0,
                  "movement-stop failure blocks buff input");
            Reset(); controller.Reset(); now = 0; ClickAimInput.F1Works = false;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 1 && ClickAimInput.Keys[0] == 0x70, "unconfirmed F1 blocks every buff");
            ClickAimInput.Keys.Clear(); now = 5000;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "failed F1 uses a longer retry backoff");
            now = 30000; Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 1, "failed F1 can retry later");
            Reset(); controller.Reset(); now = 0; ClickAimInput.StopOnF1 = true;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 1, "STOP after self-target blocks buff key");
            Reset(); controller.Reset(); now = 0; ClickAimInput.CombatAfterFirst = true;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 2, "combat after first buff blocks remaining buffs");
            GrindRunner.Combat = false; ClickAimInput.CombatAfterFirst = false; ClickAimInput.Keys.Clear(); now = 5000;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 2 && ClickAimInput.Keys[0] == 0x32, "unattempted buffs remain due");
            Reset(); controller.Reset(); now = 0; ClickAimInput.StopAfterFirst = true;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 2, "STOP after first buff blocks remaining keys");
            Reset(); controller.Reset(); now = 0; ClickAimInput.DisableAfterFirst = true;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 2, "disabling buffs after first key blocks the remainder");
            Reset(); controller.Reset(); now = 0; ClickAimInput.ChangeTargetAfterFirst = true;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 2, "changed self target blocks next buff");
            Reset(); controller.Reset(); now = 0; ClickAimInput.ChangeMapAfterFirst = true;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 2, "map change blocks next buff");
            Reset(); controller.Reset(); now = 0; ClickAimInput.ChangeSecondAfterFirst = true;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 2, "changed slot blocks stale key");
            Reset(); controller.Reset(); now = 0; WowBGrindFactions.Ghost = true;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "ghost blocks buffs");
            Reset(); controller.Reset(); now = 0; ObjectManager.WowB!.Power = 0;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "zero 70707 mana defers buff attempts");
            ObjectManager.WowB.Power = 100; now = 5000;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 4, "buffs remain due when mana recovers");
            Reset(); controller.Reset(); now = 0; ObjectManager.WowB!.IsExperimental70707 = false; ObjectManager.WowB.Power = 0;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement), "unknown 70291 power type is not treated as mana");
            Reset(); controller.Reset(); now = 0; WowBGrindFactions.GhostReadable = false;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "unreadable ghost flag blocks 70707 buffs");
            Reset(); controller.Reset(); now = 0; ObjectManager.WowB!.IsExperimental70707 = false; WowBGrindFactions.GhostReadable = false;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 4, "70291 does not require the 70707 ghost flag");
            Reset(); controller.Reset(); now = 0; WowBGrindFactions.BarSafe = false;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "unconfirmed action bar blocks buffs");
            Reset(); controller.Reset(); now = 0; ClickAimSettings.UseBuffs = false;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "disabled setting blocks buffs");
            ClickAimSettings.UseBuffs = true;
            Check(controller.TryApply(new IntPtr(7), stopped, stopMovement), "enabled during session attempts due buffs");
            Reset(); controller.Reset(); now = 0; ClickAimSettings.RotationClass = ClickAimRotationClass.Druid;
            ClickAimCombat.Bar = new List<ClickAimAction> { Action(4, 1), Action(20550, 2) };
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "implicit druid forms and passive racial excluded");
            controller.Reset(); now = 0; ClickAimSettings.RotationClass = ClickAimRotationClass.Generic;
            Check(!controller.TryApply(new IntPtr(7), stopped, stopMovement) && ClickAimInput.Keys.Count == 0, "forms excluded even if class is Generic");
            Console.WriteLine("Buff scheduling and safety fixtures passed (synthetic only; no real cast or aura validation).");
        }
    }
}

    sealed class Memory
    {
        public IntPtr WindowHandle = new IntPtr(7);
        public byte[]? AuraBytes;
        public bool ChangeSecondRead, StopDuringRead, ChangeTotemCount;
        public uint TotemCount;
        private int _countReads;
        private int _readCount;
        public byte[]? ReadBytes64(ulong address, int length)
        {
            if (address == 0x20000 + 0x1A60 && length == 4)
                return BitConverter.GetBytes(ChangeTotemCount && ++_countReads % 2 == 0 ? TotemCount + 1 : TotemCount);
            if (AuraBytes == null || address != 0x20000 + 0x5F0 || length != AuraBytes.Length) return null;
            var result = (byte[])AuraBytes.Clone();
            if (StopDuringRead) Styx.Logic.Pathing.ClickAimAbort.Requested = true;
            if (ChangeSecondRead && ++_readCount % 2 == 0) result[0xF08 - 0x5F0] ^= 1;
            return result;
        }
        public IDisposable TemporaryCacheState(bool value) => new Scope();
        private sealed class Scope : IDisposable { public void Dispose() { } }
    }

    readonly struct WowBGuid : IEquatable<WowBGuid>
    {
        private readonly int value;
        public WowBGuid(int value) { this.value = value; }
        public bool IsEmpty => value == 0;
        public bool Equals(WowBGuid other) => value == other.value;
        public override bool Equals(object? other) => other is WowBGuid guid && Equals(guid);
        public override int GetHashCode() => value;
        public static bool operator ==(WowBGuid a, WowBGuid b) => a.Equals(b);
        public static bool operator !=(WowBGuid a, WowBGuid b) => !a.Equals(b);
    }

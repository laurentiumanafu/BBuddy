using System;
using System.Collections.Generic;
using Styx.Logic.Pathing;

static class Fixture
{
    static void Check(bool value, string name) { if (!value) throw new Exception(name); }
    public static void Main()
    {
        int polls = 0; bool stop = false; bool alive;
        var logs = new List<string>();
        bool result = WowBHotspotDeathWatch.CheckOnce(
            () => { polls++; return (false, "alive candidate"); },
            () => stop, logs.Add, out alive);
        Check(!result && alive && polls == 1 && logs.Count == 1, "alive candidate is checked once and re-plans");
        polls = 0; stop = false; logs.Clear();
        result = WowBHotspotDeathWatch.CheckOnce(
            () => { polls++; return (true, "zero-health checkpoint recorded"); },
            () => stop, logs.Add, out alive);
        Check(result && !alive && polls == 1, "confirmed checkpoint returns ready on the single check");
        polls = 0; stop = false; logs.Clear();
        result = WowBHotspotDeathWatch.CheckOnce(
            () => { polls++; stop = true; return (true, "checkpoint"); },
            () => stop, logs.Add, out alive);
        Check(!result && !alive && polls == 1, "STOP during poll wins over checkpoint");
        polls = 0; stop = true; logs.Clear();
        result = WowBHotspotDeathWatch.CheckOnce(
            () => { polls++; return (true, "checkpoint"); },
            () => stop, logs.Add, out alive);
        Check(!result && !alive && polls == 0, "STOP before check");
        polls = 0; stop = false; logs.Clear();
        result = WowBHotspotDeathWatch.CheckOnce(
            () => { polls++; return (false, "unreadable"); },
            () => stop, logs.Add, out alive);
        Check(!result && !alive && polls == 1 && logs.Count == 1, "unknown is checked once and stops without re-planning");
        Console.WriteLine("Hotspot death watch fixtures passed (synthetic state only).");
    }
}

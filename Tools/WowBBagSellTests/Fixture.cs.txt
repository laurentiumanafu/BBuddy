    readonly struct WowBGuid
    {
        private readonly int _value;
        public WowBGuid(int value) { _value = value; }
        public bool IsEmpty => _value == 0;
        public static bool operator ==(WowBGuid a, WowBGuid b) => a._value == b._value;
        public static bool operator !=(WowBGuid a, WowBGuid b) => !(a == b);
        public override bool Equals(object? obj) => obj is WowBGuid other && this == other;
        public override int GetHashCode() => _value;
    }
